
cmake_minimum_required(VERSION 3.20)

project(MossGodot
    VERSION 0.1.0
    DESCRIPTION "Moss Framework integration for Godot"
    LANGUAGES CXX
)

include(FetchContent)

# ------------------------------------------------------------
# Configuration
# ------------------------------------------------------------

set(GODOTCPP_API_VERSION "4.7" CACHE STRING
    "Godot API version targeted by this GDExtension"
)

set(GODOTCPP_TARGET "template_debug" CACHE STRING
    "godot-cpp target: template_debug or template_release"
)

set(GODOTCPP_PRECISION "single" CACHE STRING
    "Godot floating point precision"
)

option(MOSS_GODOT_ENABLE_SANITIZERS
    "Enable AddressSanitizer and UndefinedBehaviorSanitizer"
    OFF
)

option(MOSS_GODOT_WARNINGS_AS_ERRORS
    "Treat MossGodot warnings as errors"
    OFF
)

# Produce a universal macOS library by default.
if(APPLE AND NOT DEFINED CMAKE_OSX_ARCHITECTURES)
    set(CMAKE_OSX_ARCHITECTURES "arm64;x86_64"
        CACHE STRING "macOS target architectures" FORCE
    )
endif()

# ------------------------------------------------------------
# godot-cpp
# ------------------------------------------------------------

FetchContent_Declare(
    godot-cpp
    GIT_REPOSITORY https://github.com/godotengine/godot-cpp.git
    GIT_TAG 10.0.0-stable
    GIT_SHALLOW TRUE
)

FetchContent_MakeAvailable(godot-cpp)

# ------------------------------------------------------------
# Sources
# ------------------------------------------------------------

add_library(moss_godot SHARED

    src/register_types.cpp

    src/subsystems/moss_subsystem.cpp
    src/subsystems/moss_subsystems.cpp

    src/physics/moss_physics_server_3d.cpp
)

target_compile_features(moss_godot PRIVATE cxx_std_17)

target_include_directories(moss_godot PRIVATE
    "${CMAKE_CURRENT_SOURCE_DIR}/src"
)

target_link_libraries(moss_godot PRIVATE
    godot-cpp
)

# ------------------------------------------------------------
# Output naming
# ------------------------------------------------------------

# Match the filename suffix generated by godot-cpp.
get_target_property(
    MOSS_GODOTCPP_SUFFIX
    godot-cpp
    GODOTCPP_SUFFIX
)

if(NOT MOSS_GODOTCPP_SUFFIX)
    message(FATAL_ERROR
        "Could not obtain the godot-cpp platform suffix."
    )
endif()

set_target_properties(moss_godot PROPERTIES

    OUTPUT_NAME "moss_godot${MOSS_GODOTCPP_SUFFIX}"

    CXX_STANDARD 17
    CXX_STANDARD_REQUIRED YES
    CXX_EXTENSIONS NO

    POSITION_INDEPENDENT_CODE ON

    LIBRARY_OUTPUT_DIRECTORY
        "${CMAKE_CURRENT_SOURCE_DIR}/addons/moss_godot/bin"

    RUNTIME_OUTPUT_DIRECTORY
        "${CMAKE_CURRENT_SOURCE_DIR}/addons/moss_godot/bin"

    ARCHIVE_OUTPUT_DIRECTORY
        "${CMAKE_BINARY_DIR}/lib"
)

# Ensure multi-configuration generators place binaries in the same folder.
foreach(CONFIG_NAME DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)

    set_target_properties(moss_godot PROPERTIES

        LIBRARY_OUTPUT_DIRECTORY_${CONFIG_NAME}
            "${CMAKE_CURRENT_SOURCE_DIR}/addons/moss_godot/bin"

        RUNTIME_OUTPUT_DIRECTORY_${CONFIG_NAME}
            "${CMAKE_CURRENT_SOURCE_DIR}/addons/moss_godot/bin"
    )

endforeach()

# ------------------------------------------------------------
# Compiler warnings
# ------------------------------------------------------------

if(MSVC)

    target_compile_options(moss_godot PRIVATE /W4)

    if(MOSS_GODOT_WARNINGS_AS_ERRORS)
        target_compile_options(moss_godot PRIVATE /WX)
    endif()

else()

    target_compile_options(moss_godot PRIVATE
        -Wall
        -Wextra
        -Wpedantic
    )

    if(MOSS_GODOT_WARNINGS_AS_ERRORS)
        target_compile_options(moss_godot PRIVATE -Werror)
    endif()

endif()

# ------------------------------------------------------------
# Sanitizers
# ------------------------------------------------------------

if(MOSS_GODOT_ENABLE_SANITIZERS)

    if(MSVC)
        message(FATAL_ERROR
            "Sanitizer configuration currently supports GCC and Clang."
        )
    endif()

    target_compile_options(moss_godot PRIVATE
        -fsanitize=address,undefined
        -fno-omit-frame-pointer
    )

    target_link_options(moss_godot PRIVATE
        -fsanitize=address,undefined
    )

endif()

message(STATUS "MossGodot version: ${PROJECT_VERSION}")
message(STATUS "Godot API: ${GODOTCPP_API_VERSION}")
message(STATUS "godot-cpp target: ${GODOTCPP_TARGET}")
message(STATUS "godot-cpp suffix: ${MOSS_GODOTCPP_SUFFIX}")
